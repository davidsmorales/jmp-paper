\begin{frame}{Previous Discussion}

\textbf{Is the exogenous death probability, $\lambda_s$, necessary on top of endogenous exit?} Yes!

\bigskip
\begin{itemize}
    \item Productive (big/old) firms will never optimally exit: 
    \begin{itemize}
    \vspace{2mm}
        \item exit iff Continuation $<$ scrap, never the case, for any admissible operating cost.
        \vspace{2mm}
        \item Formally: $\lim\limits_{\nu\to\infty}Prob(\text{choice exit}\mid \nu) = 0$, so the total exit rate tends to $\lambda_s$.
        \vspace{2mm}
        \item So, the exogenous death = Non-productivity related death. 
        \vspace{2mm}
        \item $\lambda_s$ estimated outside the model with s.e. propagated through the chain. 
        \vspace{2mm}
    \end{itemize}
\end{itemize}

\end{frame}

\begin{frame}[noframenumbering]{Formal identification argument}

Given the 9 institutional parameters and the 12 parameters estimated outside the model, the moment map $m:\Theta =\prod_{j}[\underline\theta_{j},\overline\theta_{j}] \to\mathbb{R}^{10}$ fulfills the following properties:

\begin{itemize}
    \item \textbf{Relevance}: Each moment is strictly monotone in its own parameter, $\frac{\partial m_j}{\partial \theta_j} \neq 0$.
    \item \textbf{Exclusion}: $\frac{\partial \text{threshold}_{\text{formality = s}}}{\partial \text{Financial conditions}_{\text{formality = s}}} > \frac{\partial \text{threshold}_{\text{formality = s}}}{\partial \text{Financial conditions}_{\text{formality = s'}}}$, at every $\theta$ in the box.
\end{itemize}
\vspace{4mm}

\textbf{Verifiable assumption}: The elasticity Jacobian $E(\theta)$, $E_{jk}=\partial\ln m_{j}/\partial\ln\theta_{k}$, has
positive diagonal and is strictly row diagonally
dominant: $E_{jj}>\sum_{k\neq j}|E_{jk}|$ $\forall j$.
\vspace{4mm}

\textbf{Point identification}: Under the previous assumption, following standard arguments (Gale-Nikaidô, 1965; Berry-Gandhi-Haile, 2013), the moment map $m$ is univalent on $\Theta$:
\begin{itemize}
    \item For any interior data vector $\hat m\in m(\Theta)$ there is exactly one $\theta\in\Theta$ with $m(\theta)=\hat m$.
    \item The damped Gauss--Seidel iteration solving the system is a contraction.
\end{itemize}
\vspace{4mm}

\textbf{Evidence}: 
\begin{itemize}
    \item Jacobian heat map (dominance and sign of derivatives). 
    \item Multi-start, convergence from different initial conditions. 
\end{itemize}

\end{frame}

\begin{frame}[noframenumbering]{Distribution of the estimator}

Given the out of the model estimates variance, $V_{\gamma}$, I solve the ten-equation system $m(\theta;\widehat\gamma)=\hat m$ exactly, invariant to weighting. Then, 
\begin{itemize}
    \item The sampling variation reaches $\widehat\theta$ through two channels: the targeted moments and the pre-estimated inputs. Expanding $m(\widehat\theta;\widehat\gamma)=\hat m$ around $(\theta_{0},\gamma_{0})$,
\end{itemize}

\begin{equation*}
  \widehat{\mathrm{Var}}(\widehat\theta)
  \;=\;
  \hat J_{\theta}^{-1}\,
  \widehat{\mathrm{Var}}\!\big(\hat m - \hat J_{\gamma}\,
  \widehat\gamma\big)\,
  \hat J_{\theta}^{-1\prime},
  \qquad
  \hat J_{\theta}=\frac{\partial m}{\partial\theta'},\quad
  \hat J_{\gamma}=\frac{\partial m}{\partial\gamma'},
  \label{eq:id-var}
\end{equation*}

\begin{itemize}
    \item The inner variance expands into $V_{\hat m}+\hat J_{\gamma}V_{\gamma}\hat J_{\gamma}' -2\,\mathrm{Sym}\{\mathrm{Cov}(\hat m,\widehat\gamma\,') \hat J_{\gamma}'\}$
   \begin{itemize}
       \item $V_{\hat m}$ is the usual moment-sampling term.
       \item $\hat J_{\gamma}V_{\gamma}\hat J_{\gamma}'$ propagates the out of the model estimates sampling error.
       \item $2\,\mathrm{Sym}\{\mathrm{Cov}(\hat m,\widehat\gamma\,') \hat J_{\gamma}'\}$ accounts for the fact that $\hat m$ and $\widehat\gamma$ are computed from same census.
       \begin{itemize}
           \item Bootstrapped: Resample census at the market level and ENAFIN surveys, recompute $(\hat m,\widehat\gamma)$ on each replication, and re-solve system of equations at $\widehat\theta$.
       \end{itemize}
   \end{itemize}
\end{itemize}

\end{frame}
